\section{Introduzione al Progetto}

\begin{frame}{Cos'è DietiEstates?}
    \begin{block}{Obiettivo del Progetto}
        \textbf{DietiEstates} è un portale immobiliare full-stack progettato per modernizzare l'interazione tra clienti, agenti immobiliari e gestori di agenzie.
        \vspace{1em}

        L'obiettivo è creare una piattaforma:
        \begin{itemize}
            \item \textbf{Performante:} Tempi di risposta rapidi e ricerca efficiente.
            \item \textbf{Intuitiva:} Esperienza utente pulita e facile da usare.
            \item \textbf{Sicura:} Autenticazione robusta e protezione dei dati.
            \item \textbf{Modulare:} Architettura manutenibile e scalabile.
        \end{itemize}
    \end{block}
\end{frame}

\section{Architettura e Design}

\begin{frame}{Architettura del Sistema: Three-Tier}
    Il sistema segue un'architettura a tre livelli per garantire modularità e disaccoppiamento.
    \vspace{1em}

    \begin{block}{Presentation Tier (Frontend)}
        \begin{itemize}
            \item \textbf{Single-Page Application (SPA)} sviluppata in \textbf{Angular}.
            \item Responsabile dell'interfaccia utente e delle interazioni.
            \item Comunica con il backend tramite API REST.
        \end{itemize}
    \end{block}

    \begin{block}{Application Tier (Backend)}
        \begin{itemize}
            \item Applicazione \textbf{Spring Boot} che espone API RESTful.
            \item Implementa la logica di business, l'autenticazione e l'autorizzazione.
        \end{itemize}
    \end{block}

    \begin{block}{Data Tier (Database)}
        \begin{itemize}
            \item Database relazionale \textbf{PostgreSQL} per la persistenza dei dati.
            \item Accesso gestito tramite \textbf{Spring Data JPA} e Hibernate.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Tecnologie Adottate}
    \begin{columns}[T]
        \begin{column}{.5\textwidth}
            \begin{block}{Backend}
                \begin{itemize}
                    \item \textbf{Java 17} e \textbf{Spring Boot}
                    \item \textbf{Spring Security} (JWT, OIDC)
                    \item \textbf{JPA / Hibernate}
                    \item \textbf{PostgreSQL + PostGIS}
                    \item \textbf{Maven}
                    \item \textbf{Docker}
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{.5\textwidth}
            \begin{block}{Frontend}
                \begin{itemize}
                    \item \textbf{Angular} e \textbf{TypeScript}
                    \item \textbf{OpenAPI Generator}
                    \item HTML5 / CSS3 (Bootstrap)
                    \item Leaflet.js (Mappe)
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
    \begin{block}{Integrazioni Esterne}
        \begin{itemize}
            \item \textbf{Amazon S3:} per lo storage di oggetti (immagini degli annunci).
            \item \textbf{Provider OIDC} (es. Google) per l'autenticazione esterna.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Backend Design: Monolite Modulare (DDD)}
    \begin{block}{Approccio}
        Il backend è un \textbf{monolite modulare} progettato secondo i principi del \textbf{Domain-Driven Design (DDD)}.
        \begin{itemize}
            \item Ogni package Java è un \textbf{Bounded Context} con responsabilità chiare e confini ben definiti.
            \item Favorisce separazione di responsabilità: Domain, Repository, Service, Controller, DTO/Mapper.
        \end{itemize}
    \end{block}
    \begin{block}{Bounded Contexts Principali}
        \begin{itemize}
            \item \texttt{auth}: gestione utenti, ruoli, refresh token e OTP.
            \item \texttt{listings}: dominio degli annunci, ricerche geospaziali e immagini.
            \item \texttt{agency}: gestione agenzie e promozione agenti.
            \item \texttt{file}/\texttt{storage}: integrazione con S3 per contenuti binari.
            \item \texttt{email}: notifiche e template.
            \item \texttt{security}: filtri, JWT e integrazione OIDC.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Scelte di Persistenza e Schema DB}
    \begin{itemize}
        \item Database relazionale: \textbf{PostgreSQL} per consistenza, transazioni e maturità.
        \item Geodati: estensione \textbf{PostGIS} — campi \texttt{geometry(Point,4326)} e calcoli geodetici con \texttt{geography} per distanze in metri.
        \item Indici spaziali GIST su \texttt{listing.coordinate} per ricerche per prossimità performanti.
        \item Separazione dati binari/metadati: file su S3, metadati (URL, sort\_order, timestamps) in \texttt{listing\_image}.
        \item Funzioni/stored procedures per operazioni atomiche (es. revoke\_agent, get\_listings\_by\_agent).
    \end{itemize}
\end{frame}

\begin{frame}{Motivazioni per PostGIS e Query Geospaziali}
    \begin{itemize}
        \item PostGIS offre operatori comprovati: \texttt{ST\_DWithin}, \texttt{ST\_Distance}, \texttt{ST\_Transform} e funzioni di estrazione \texttt{ST\_X}/\texttt{ST\_Y}.
        \item Indici GIST riducono rapidamente lo spazio di ricerca prima dei calcoli costosi.
        \item Utilizzo di \texttt{geography} per calcoli in metri e accuratezza geodetica.
        \item Query ottimizzate restituendo anche \texttt{thumbnail\_url} e coordinate per la UI.
    \end{itemize}
\end{frame}

\begin{frame}{Design e Pattern Architetturali}
    \begin{itemize}
        \item \textbf{Factory Pattern} per OIDC/OAuth2 user info: \texttt{OAuth2UserInfoFactory} isola mapping provider-specific.
        \item \textbf{Security Filter Chain}: pipeline di filtri Spring Security con filtro custom (es. \texttt{AuthFilter}) per validazione JWT e popolamento del \texttt{SecurityContext}.
        \item Uso di DTO/Mapper per separare API contract da model di dominio e facilitare test.
    \end{itemize}
\end{frame}

\begin{frame}{Sicurezza: OIDC, JWT e Refresh Tokens}
    \begin{itemize}
        \item Autenticazione delegata a provider OIDC (es. Google) per sicurezza e UX unificata.
        \item Access token: \textbf{JWT} stateless per autorizzare richieste API.
        \item Refresh token: persistiti e gestiti per rinnovare access token senza re-login.
        \item Policy di autorizzazione basate sui ruoli (ROLE\_ADMIN, ROLE\_AGENT, ROLE\_USER) definite nella \texttt{SecurityFilterChain}.
    \end{itemize}
\end{frame}

\begin{frame}{Frontend: Contract-First e Strumenti}
    \begin{itemize}
        \item SPA con \textbf{Angular} e TypeScript per UX reattiva e component-based design.
        \item \textbf{OpenAPI Generator} per generare client TypeScript type-safe dal contratto backend: riduce disallineamenti e velocizza integrazione.
        \item Design responsive mobile-first con componenti riutilizzabili (cards, form, map view).
    \end{itemize}
\end{frame}

\section{Approfondimenti Tecnici}

\begin{frame}[fragile]{Deep Dive: Ricerca Geospaziale con PostGIS}
    \begin{block}{Problema}
        Permettere agli utenti di cercare immobili entro un raggio specifico da un punto sulla mappa.
    \end{block}
    \begin{block}{Soluzione}
        \begin{itemize}
            \item Utilizzo dell'estensione \textbf{PostGIS} per PostgreSQL.
            \item Le coordinate degli immobili sono salvate in un campo di tipo \texttt{geometry(Point, 4326)}.
            \item Le query sfruttano indici spaziali (GIST) e funzioni native per calcoli geodetici efficienti.
        \end{itemize}
    \end{block}
    \begin{exampleblock}{Esempio di Query di ricerca per raggio}
\begin{lstlisting}[language=SQL]
SELECT l.* FROM listing l
WHERE ST_DWithin(
    l.coordinate::geography,
    ST_SetSRID(ST_MakePoint(:lon, :lat), 4326)::geography,
    :radius_meters
);
\end{lstlisting}
    \end{exampleblock}
\end{frame}

\begin{frame}{Deep Dive: Gestione della Sicurezza}
    \begin{block}{Autenticazione e Autorizzazione}
        La sicurezza è gestita da \textbf{Spring Security} con un approccio \textit{stateless}.
        \begin{itemize}
            \item \textbf{Autenticazione via Credenziali:} Generazione di \textbf{JWT Access Token} e \textbf{Refresh Token}.
            \item \textbf{Autenticazione via Provider Esterni:} Supporto a \textbf{OpenID Connect (OIDC)} (es. Google) tramite il pattern \textit{OAuth2UserInfoFactory}.
            \item \textbf{Autorizzazione:} Gestita tramite una \texttt{SecurityFilterChain} che protegge gli endpoint in base ai ruoli (\texttt{ADMIN}, \texttt{AGENT}, \texttt{USER}).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Deep Dive: Gestione delle Immagini}
    \begin{block}{Separazione delle Responsabilità}
        Per garantire scalabilità e performance, i dati binari sono separati dai metadati.
        \begin{itemize}
            \item \textbf{Amazon S3:} Utilizzato come object storage per salvare i file delle immagini. È scalabile, robusto e ottimizzato per la distribuzione di contenuti.
            \item \textbf{PostgreSQL:} Memorizza solo i metadati associati alle immagini (URL, ordine di visualizzazione, riferimento all'annuncio).
        \end{itemize}
    \end{block}
    \begin{alertblock}{Vantaggi}
        \begin{itemize}
            \item Il database rimane leggero e performante.
            \item Lo storage degli asset può scalare indipendentemente.
        \end{itemize}
    \end{alertblock}
\end{frame}
